import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats
pd.set_option("display.width", 85)

def analyse(data, xcol, ycol, xlabel, ylabel, title):
    y = data[ycol]
    print("Statistics of", ylabel)
    print("Mean               :", round(y.mean(), 2))
    print("Median             :", round(y.median(), 2))
    print("Variance (sample) :", round(y.var(), 2))
    print("Standard deviation :", round(y.std(), 2))
    r = data[xcol].corr(y)
    print("Correlation (%s vs %s): %.4f" % (xlabel, ylabel, r))

     fig, ax = plt.subplots(1, 2, figsize=(10, 3.8))
     ax[0].scatter(data[xcol], y, color="steelblue", edgecolor="black")
     m, c = np.polyfit(data[xcol], y, 1)
     ax[0].plot(data[xcol], m * data[xcol] + c, color="red")
     ax[0].set_xlabel(xlabel); ax[0].set_ylabel(ylabel)
     ax[0].set_title("Scatter plot: %s vs %s" % (xlabel, ylabel))
     ax[1].hist(y, bins=8, density=True, color="lightgray", edgecolor="black")
     xs = np.linspace(y.min(), y.max(), 200)
     ax[1].plot(xs, stats.norm.pdf(xs, y.mean(), y.std()), color="red", linewidth=2)
     ax[1].set_xlabel(ylabel); ax[1].set_ylabel("Density")
     ax[1].set_title("Histogram with normal curve")
     fig.suptitle(title)
     plt.tight_layout()
     plt.show()
     return r

def interpret(r, y, xname):
    if abs(r) >= 0.7: s = "strong"
    elif abs(r) >= 0.4: s = "moderate"
    else: s = "weak"
    d = "positive" if r > 0 else "negative"
    print("Interpretation:")
    print(" - Correlation r = %.2f, a %s %s relationship with %s." % (r, s, d, xname))
    print(" - Skewness of data = %.2f (%s)." % (stats.skew(y),
          "approximately symmetric" if abs(stats.skew(y)) < 0.5 else "skewed"))

Program section 2:
class SimplePerceptron:
   """Single neuron with a step activation, trained with the perceptron learning rule."""
   def __init__(self, lr=0.1, epochs=100):
       self.lr = lr
       self.epochs = epochs

   def step(self, z):
       return np.where(z >= 0, 1, 0)

   def predict(self, X):
       return self.step(X @ self.w + self.b)

   def fit(self, X, y, verbose=True):
       self.w = np.zeros(X.shape[1])
       self.b = 0.0
       for epoch in range(1, self.epochs + 1):
           errors = 0
           for xi, target in zip(X, y):
               out = self.step(xi @ self.w + self.b)
               update = self.lr * (target - out)

def sigmoid(z):
  return 1 / (1 + np.exp(-z))

 rng = np.random.default_rng(0)
 hidden_units, lr = 4, 0.5
 W1 = rng.normal(0, 1, (2, hidden_units)); b1 = np.zeros(hidden_units)
 W2 = rng.normal(0, 1, (hidden_units, 1)); b2 = np.zeros(1)
 t = y_xor.reshape(-1, 1)
 loss_history = []

 for epoch in range(10001):
  a1 = sigmoid(X @ W1 + b1)                 # forward pass: hidden layer
  a2 = sigmoid(a1 @ W2 + b2)                # forward pass: output layer
  loss = np.mean((t - a2) ** 2)
  loss_history.append(loss)
  d2 = (a2 - t) * a2 * (1 - a2)             # backward pass: output error
  d1 = (d2 @ W2.T) * a1 * (1 - a1)          # backward pass: hidden error
  W2 -= lr * a1.T @ d2; b2 -= lr * d2.sum(axis=0)
  W1 -= lr * X.T @ d1; b1 -= lr * d1.sum(axis=0)
  if epoch % 2000 == 0:
